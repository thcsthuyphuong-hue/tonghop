match /exam_rooms/{roomId} {
  function isExamHost() {
    return request.auth != null
      && request.auth.uid == "wg7QT6isC2dryLf8rxc6z9lUjNd2"
      && request.auth.token.firebase.sign_in_provider == "password";
  }
  allow read: if request.auth != null;
  allow create: if isExamHost()
    && request.resource.data.hostUid == request.auth.uid
    && request.resource.data.status == "waiting";
  allow update: if isExamHost()
    && resource.data.hostUid == request.auth.uid
    && request.resource.data.hostUid == resource.data.hostUid
    && request.resource.data.diff(resource.data).affectedKeys()
      .hasOnly(["status", "readingStartedAt", "examEndsAt", "finishedAt"]);
  allow delete: if false;

  match /participants/{uid} {
    function room() {
      return get(/databases/$(database)/documents/exam_rooms/$(roomId)).data;
    }
    function ownsRoom() {
      return isExamHost() && room().hostUid == request.auth.uid;
    }
    // Host cần quyền list trên toàn bộ collection để onSnapshot(collection(...)) hoạt động.
    // Học sinh chỉ được get hồ sơ của chính mình.
    allow get: if request.auth != null && (request.auth.uid == uid || ownsRoom());
    allow list: if ownsRoom();
    allow create: if request.auth != null
      && request.auth.uid == uid
      && request.resource.data.uid == uid
      && request.resource.data.approved == false
      && request.resource.data.submitted == false
      && (room().status == "waiting"
        || (room().status == "reading"
          && request.time < room().readingStartedAt + duration.value(5, "m")));
    allow update: if (ownsRoom()
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["approved", "approvedAt"]))
      || (request.auth != null && request.auth.uid == uid
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(["name", "class", "sbd", "school"]));
    allow delete: if false;
  }

  match /submissions/{uid} {
    function room() {
      return get(/databases/$(database)/documents/exam_rooms/$(roomId)).data;
    }
    function ownsRoom() {
      return isExamHost() && room().hostUid == request.auth.uid;
    }
    function approvedParticipant() {
      return get(/databases/$(database)/documents/exam_rooms/$(roomId)/participants/$(uid)).data.approved == true;
    }
    allow read: if request.auth != null && (request.auth.uid == uid || ownsRoom());
    allow create: if request.auth != null
      && request.auth.uid == uid
      && approvedParticipant()
      && room().status == "reading"
      && room().readingStartedAt != null
      && room().examEndsAt != null
      && request.time >= room().readingStartedAt + duration.value(5, "m")
      && request.time < room().examEndsAt
      && request.resource.data.uid == uid
      && request.resource.data.state == "submitted"
      && request.resource.data.submittedAt == request.time
      && request.resource.data.keys().hasOnly(["uid", "answers", "submittedAt", "state"]);
    allow update: if ownsRoom()
      && request.resource.data.diff(resource.data).affectedKeys()
        .hasOnly(["manualShortScore", "manualScores", "gradeState", "finalScore"]);
    allow delete: if false;
  }

  match /private/{privateDocId} {
    allow read, write: if isExamHost()
      && get(/databases/$(database)/documents/exam_rooms/$(roomId)).data.hostUid == request.auth.uid;
  }

  match /grades/{uid} {
    allow read, write: if isExamHost()
      && get(/databases/$(database)/documents/exam_rooms/$(roomId)).data.hostUid == request.auth.uid;
  }
}
